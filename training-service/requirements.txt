#
#   TRAINING_SERVICE_TOKEN=<shared secret> \
#   DATA_ROOT=/var/lib/training \
#   LLAMAFACTORY_HOME=/opt/LLaMA-Factory \
#   TRAINING_SERVICE_HOST=0.0.0.0 \
#   uvicorn app.main:app --host "$TRAINING_SERVICE_HOST" --port 8083
